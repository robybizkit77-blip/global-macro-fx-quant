  const a=NATIVE_RATES_DATA?.[c]?.history2;

  const u=NATIVE_RATES_DATA?.USD?.history2;
  if(!a?.dates?.length||!u?.dates?.length) return {labels:[],values:[],note:'WITHHELD: storico 2Y non disponibile per una delle due gambe.'};
  const um=new Map((u.dates||[]).map((d,i)=>[d,Number(u.values?.[i])]));
  const labels=[],values=[];
  (a.dates||[]).forEach((d,i)=>{
    const av=Number(a.values?.[i]),uv=um.get(d);
    if(Number.isFinite(av)&&Number.isFinite(uv)){labels.push(d);values.push((av-uv)*100);}
  });
  const lim=260;
  if(values.length<2) return {labels:[],values:[],note:'WITHHELD: date comuni insufficienti per costruire il differenziale 2Y.'};
  const note=NATIVE_RATES_DATA?.[c]?.history_status==='SEPARATE_BASIS_CURRENT_SNAPSHOT_NOT_SPLICED'
    ? 'Storico 2Y mostrato sulla propria base continua; lo snapshot corrente resta separato finché la continuità non è validata.'
    : 'Differenziale storico 2Y della valuta meno USD, in punti base.';
  return {labels:labels.slice(-lim),values:values.slice(-lim),note};
}

function newsPulseFreshness(){
  const raw=window.AUXILIARY_FEED_FRESHNESS?.news?.as_of || null;
  const status=window.AUXILIARY_FEED_FRESHNESS?.news?.status || 'UNKNOWN';
  if(!raw) return {fresh:false,asOf:null,status,reason:'timestamp news non disponibile'};
  const d=new Date(raw);
  if(!Number.isFinite(d.getTime())) return {fresh:false,asOf:raw,status,reason:'timestamp news non interpretabile'};
  const now=new Date();
  const localDate=x=>[
    x.getFullYear(),
    String(x.getMonth()+1).padStart(2,'0'),
    String(x.getDate()).padStart(2,'0')
  ].join('-');
  const fresh=localDate(d)===localDate(now);
  return {
    fresh,
    asOf:raw,
    status,
    reason:fresh?'feed validato oggi':'ultimo feed validato in una giornata precedente'
  };
}
