function v256RenderRatesRank(){
const host=document.querySelector('#v256RatesRank');if(!host)return;
const ccys=['USD','EUR','GBP','JPY','CHF','CAD','AUD','NZD'];
function historyWeeklyCheck(c,d){
const stated=Number(d?.chg2_bp);
if(!Number.isFinite(stated)) return {ok:null,note:'variazione 1W WITHHELD'};
const h=d?.history2;
if(!h || !Array.isArray(h.dates) || !Array.isArray(h.values) || h.dates.length<2) return {ok:null,note:'storico non sufficiente'};
const endDate=String(d.date||'');
const endIdx=h.dates.lastIndexOf(endDate);
if(endIdx<0) return {ok:null,note:'snapshot corrente non ancora sincronizzato nello storico'};
const endVal=Number(h.values[endIdx]);
const endMs=Date.parse(endDate.replaceAll('/','-'));
if(!Number.isFinite(endVal)||!Number.isFinite(endMs)) return {ok:null,note:'storico non confrontabile'};
let best=-1,bestGap=Infinity;
for(let i=0;i<endIdx;i++){
const ms=Date.parse(String(h.dates[i]).replaceAll('/','-'));
if(!Number.isFinite(ms)) continue;
const days=(endMs-ms)/86400000;
if(days<5 || days>10) continue;
const gap=Math.abs(days-7);
if(gap<bestGap){bestGap=gap;best=i;}
}
if(best<0) return {ok:null,note:'manca un punto circa 1 settimana prima'};
const startVal=Number(h.values[best]);
if(!Number.isFinite(startVal)) return {ok:null,note:'storico non confrontabile'};
const calc=(endVal-startVal)*100;
const diff=Math.abs(calc-stated);
return {
ok:diff<=1.0,calc,stated,startDate:h.dates[best],
note:diff<=1.0
? `coerente con storico ${h.dates[best]} → ${endDate}`
: `storico non sincronizzato con lo snapshot validato`
};
}
const rows=ccys.map(c=>{
const d=NATIVE_RATES_DATA?.[c]||{};
const raw=d.chg2_bp;
const v=(raw===null||raw===undefined)?null:Number(raw);
return {c,v,date:d.date||'—',source:d.source||'—',quality:d.quality||'',check:historyWeeklyCheck(c,d)};
});
const ranked=rows.filter(r=>Number.isFinite(r.v)).sort((a,b)=>b.v-a.v);
const withheld=rows.filter(r=>!Number.isFinite(r.v));
const max=Math.max(1,...ranked.map(x=>Math.abs(x.v)));
host.innerHTML=
ranked.map(r=>{
const w=Math.max(2,Math.abs(r.v)/max*49);
const cls=r.v>=0?'pos':'neg';
const integrity=r.check?.ok===false?'warn':r.check?.ok===true?'ok':'unknown';
const integrityText=r.check?.ok===false?'⚠':r.check?.ok===true?'✓':'?';
return `<div class="v256RankRow ${integrity}"><div class="v256RankCcy">${r.c}<small>${r.date}</small></div><div class="v256RankTrack"><div class="v256RankZero"></div><div class="v256RankBar ${cls}" style="width:${w}%"></div></div><div class="v256RankVal ${cls}">${r.v>=0?'+':''}${r.v.toFixed(1)} bp <small title="${r.check?.note||''}">${integrityText}</small></div></div>`;
}).join('')+
withheld.map(r=>`<div class="v256RankRow withheld"><div class="v256RankCcy">${r.c}<small>${r.date}</small></div><div class="v256RankTrack"><div class="v256RankZero"></div></div><div class="v256RankVal neu">WITHHELD</div></div>`).join('');
let note=document.getElementById('v317RatesIntegrityNote');
if(!note){
note=document.createElement('div');
note.id='v317RatesIntegrityNote';
note.className='v317RatesIntegrityNote';
host.insertAdjacentElement('afterend',note);
}
const unsynced=ranked.filter(r=>r.check?.ok===false);
const currentOnly=ranked.filter(r=>r.check?.ok===null);
note.innerHTML=`<b>Integrità Rates:</b> EUR 2Y canonico 25/09 validato; storico legacy separato perché non same-basis. GBP BoE same-basis: 2Y −1,9 bp e 10Y +9,4 bp (18→25/09).`+
(unsynced.length?` ⚠ Storico da risincronizzare: ${unsynced.map(x=>x.c).join(', ')}.`:'');
}
