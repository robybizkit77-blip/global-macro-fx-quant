function macroStateClass(label,state){
const s=(state||'').toUpperCase();
const l=(label||'').toUpperCase();
if(!s) return 'info';
if(/MIST|MIXED|NEUTRAL|DIVERG|PARZIAL/.test(s)) return 'mix';
if(/DETERIOR|WEAK|DEBOL|CONTRACT|COOLING|TIGHTEN|RALLENT/.test(s)) return 'bad';
if(l.includes('INFLAZ') && /PERSIST|REACCEL|ELEVAT|ALTA|STICKY/.test(s)) return 'warn';
if(l.includes('SALAR') && /ELEVAT|STRONG/.test(s)) return 'warn';
if(/STRONG|RESILIENT|SUPPORTIVE|FAVOREV|EXPAND|ACCELER|BENIGN|EASING|RECUPERO|SOLID/.test(s)) return 'good';
if(/PERSIST|REACCEL|COSTLY|RIGID|PRESSURE/.test(s)) return 'warn';
return 'info';
}
function macroSimpleScore(v,kind){

return 'pressione più distante dalla normalità';
}
if(v>=0.75)return 'forte';
if(v>=0.2)return 'resiliente';
if(v>-0.2)return 'mista / neutra';
if(v>-0.75)return 'debole';
return 'molto debole';
}

// Direction helpers: arrows show variable direction only, never FX judgement.
function dirArrow(curr,prev,eps=1e-12){
if(curr==null||prev==null||!isFinite(curr)||!isFinite(prev)) return '<span class="dirFlat dirArrow">→</span>';
const d=curr-prev;
if(Math.abs(d)<=eps) return '<span class="dirFlat dirArrow">→</span>';
return d>0?'<span class="dirUp dirArrow">↑</span>':'<span class="dirDown dirArrow">↓</span>';
}
function dirSymbol(curr,prev,eps=1e-12){
if(curr==null||prev==null||!isFinite(curr)||!isFinite(prev)) return '<span class="dirFlat dirArrow">→</span>';
const d=curr-prev;
if(Math.abs(d)<=eps) return '<span class="dirFlat dirArrow">→</span>';
return d>0?'<span class="dirUp dirArrow">↑</span>':'<span class="dirDown dirArrow">↓</span>';
}

function lastPair(x){
if(!x||!x.values||x.values.length<1) return [null,null];
const a=x.values.at(-1), b=x.values.length>1?x.values.at(-2):null;
return [a,b];
}
function rate2YMove(c){
const s=rate2YSeries(c), [curr,prev]=lastPair(s);
return {c,s,curr,prev,delta:(curr!=null&&prev!=null)?curr-prev:null};
}

function lastDirectionalTurn(seriesObj, lookback=30){
if(!seriesObj?.values?.length || !seriesObj?.dates?.length) return null;
const vals=seriesObj.values.map(Number);
const dates=seriesObj.dates;
if(vals.length<8) return null;
const start=Math.max(3, vals.length-lookback);
let last=null;
for(let i=start;i<vals.length;i++){
const prevDelta=vals[i-1]-vals[i-3];
const currDelta=vals[i]-vals[i-2];
if(!isFinite(prevDelta) || !isFinite(currDelta)) continue;
const prevSign=Math.sign(prevDelta);
const currSign=Math.sign(currDelta);
if(prevSign!==0 && currSign!==0 && prevSign!==currSign){
last={date:dates[i],dir:currSign,index:i};
}
}
return last;
}

function deriveLeadStructure(c){
const d=currencyIntelData?.[c];
if(!d) return {
lead:'Non determinato',
confirm:'—',
diverge:'—',
lag:'Non determinato',
overlay:'Liquidità non determinata',
leadReason:'dati insufficienti'
};
const coreLayers=[
{key:'macro',name:'Macro',pol:layerPolarity(d.macro)},
{key:'rates',name:'Rates / aspettative CB',pol:policyFamilyPolarity(d)},
{key:'price',name:'Prezzo',pol:layerPolarity(d.price)}
];
const liqPol=layerPolarity(d.liq);
const overlay =
liqPol>0 ? 'Liquidità favorevole' :
liqPol<0 ? 'Liquidità contraria' :
'Liquidità neutra';
// Lead and Lag are temporal concepts.
// At this stage only Rates and Price have directly comparable dated turning points.
const rateTurn=lastDirectionalTurn(ratesSeriesObjectForCurrency(c));
const priceTurn=lastDirectionalTurn(priceSeriesObjectForCurrency(c));
let lead='Non determinato';
let lag='Non determinato';
let leadReason='tempistiche non confrontabili con sufficiente sicurezza';
if(rateTurn && priceTurn){
const gap=daysBetweenDates(rateTurn.date,priceTurn.date);
if(gap!=null && gap>=2){
const rt=new Date(rateTurn.date).getTime();
const pt=new Date(priceTurn.date).getTime();
if(rt<pt){
lead='Rates / aspettative CB';
lag='Prezzo';
leadReason=`turning point rates ${rateTurn.date} prima del prezzo ${priceTurn.date}`;
}else if(pt<rt){
lead='Prezzo';
lag='Rates / aspettative CB';
leadReason=`turning point prezzo ${priceTurn.date} prima dei rates ${rateTurn.date}`;
}
}else if(gap!=null){
lead='Quasi simultanei';
lag='Nessun lag temporale chiaro';
leadReason='rates e prezzo hanno girato nello stesso intervallo temporale';
}
}
const driver=deriveCurrencyDriver(c);
const jointDriver=driver.startsWith('Macro + Rates');
const driverKey =
jointDriver ? null :
driver.startsWith('Macro')?'macro':
driver.startsWith('Rates')?'rates':null;
const policyPol=policyFamilyPolarity(d);
const driverPol = jointDriver
? (layerPolarity(d.macro)===policyPol ? layerPolarity(d.macro) : 0)
: driverKey==='rates'
? policyPol
: (driverKey ? coreLayers.find(x=>x.key===driverKey)?.pol : 0);
const compareLayers = jointDriver
? coreLayers.filter(x=>x.key==='price')
: coreLayers.filter(x=>x.key!==driverKey);
const confirms = driverPol
? compareLayers.filter(x=>x.pol===driverPol).map(x=>x.name)
: [];
const diverges = driverPol
? compareLayers.filter(x=>x.pol!==0 && x.pol!==driverPol).map(x=>x.name)
: [];
return {

lag,
overlay
};
}

function deriveCurrencyActionPlan(c){
const d=currencyIntelData?.[c];
if(!d) return {strengthen:'—',weaken:'—',watch:'—'};
const layers=coreDirectionalLayers(c);
const state=deriveCurrencyState(c);
const lead=deriveLeadStructure(c);
const core=layers.filter(x=>x.key!=='price');
const positive=core.filter(x=>x.pol>0);
const negative=core.filter(x=>x.pol<0);
const neutral=core.filter(x=>x.pol===0);
const price=layers.find(x=>x.key==='price');
const liqPol=layerPolarity(d.liq);
const liqTxt =
liqPol>0 ? 'liquidità favorevole' :
liqPol<0 ? 'liquidità contraria' :
'liquidità neutra';
const cotNote=cotEarlyWarning(c);
let strengthen='';
let weaken='';
if(state?.base==='FORTE'){
strengthen='Macro e Rates/aspettative CB devono restare allineati';
}else if(state?.base==='COSTRUTTIVA'){
strengthen=`Serve che ${neutral[0]?.name || 'il layer neutrale'} si allinei al layer favorevole`;
}else if(state?.base==='FRAGILE'){
strengthen='Serve un miglioramento congiunto di Macro e Rates/aspettative CB';
}else if(state?.base==='DEBOLE'){
strengthen=`Serve che ${negative[0]?.name || 'il layer contrario'} smetta di deteriorare il quadro`;
}else{
strengthen='Serve che Macro e Rates/aspettative CB convergano nella stessa direzione';
}
if(price?.pol===0){
strengthen+='; il Prezzo deve diventare più risolutivo';
}else if(state?.confirmation==='PREZZO DIVERGE'){
strengthen+='; la divergenza del Prezzo può riassorbirsi solo se il core resta valido';
}else if(state?.confirmation==='PREZZO CONFERMA'){
strengthen+='; il Prezzo è già allineato: non usarlo come prova di ulteriore continuazione';
}
if(state?.base==='FORTE' || state?.base==='COSTRUTTIVA'){
weaken='Il quadro si indebolisce se Macro o Rates/aspettative CB perdono supporto';
}else if(state?.base==='FRAGILE' || state?.base==='DEBOLE'){
weaken='La debolezza aumenta se Macro e Rates/aspettative CB restano contrari';
}else{
weaken='La lettura peggiora se Macro e Rates/aspettative CB divergono ulteriormente';
}
if(cotNote){
weaken+='; il COT può amplificare il rischio di reazione';
}
const watchParts=[];
if(lead.lead && lead.lead!=='—' && lead.lead!=='Non determinato'){
watchParts.push(`Lead: ${lead.lead}${lead.leadReason?` (${lead.leadReason})`:''}`);
}
if(lead.diverge && !lead.diverge.startsWith('Nessuna')){
watchParts.push(`Divergenza tra i segnali: ${lead.diverge}`);
}
if(lead.lag && !lead.lag.startsWith('Nessun') && lead.lag!=='Non determinato'){
watchParts.push(`Lag: ${lead.lag}`);
}
if(liqPol!==0){
watchParts.push(`Overlay: ${liqTxt}`);
}
if(cotNote){
watchParts.push(cotNote);
}
if(!watchParts.length){
watchParts.push('Macro, Rates/aspettative CB e conferma del Prezzo');
}
return {
strengthen:strengthen.charAt(0).toUpperCase()+strengthen.slice(1)+'.',
weaken:weaken.charAt(0).toUpperCase()+weaken.slice(1)+'.',
watch:watchParts.join(' · ')
};
}
function refreshActionPlanInData(c){
const d=currencyIntelData?.[c];
if(!d) return;
const p=deriveCurrencyActionPlan(c);
d.strengthen=p.strengthen;
d.weaken=p.weaken;
d.watch=p.watch;
}
function deriveCurrencyRegime(c){
const m=D.macro?.[c]||{};
const cycle=(m.cycle||'').toUpperCase();
// Prefer the canonical macro-cycle field already present in the engine.
if(cycle.includes('GOLDILOCKS')) return 'Espansione equilibrata';
if(cycle.includes('SLOWDOWN')) return 'Rallentamento';
if(cycle.includes('STAGFLATION')) return 'Stagflazione';
if(cycle.includes('REFLATION')) return 'Reflazione';
// Fallback only if canonical cycle is unavailable.
const d=currencyIntelData?.[c];
if(!d) return 'Misto';
const mp=layerPolarity(d.macro);
if(mp>0) return 'Espansione / reflazione';
if(mp<0) return 'Rallentamento';
return 'Misto';
}
function deriveCurrencyStory(c){
const d=currencyIntelData?.[c];
if(!d) return '—';
const x=deriveCurrencyState(c);
const lead=deriveLeadStructure(c);
const driver=deriveCurrencyDriver(c);
const cot=cotEarlyWarning(c);
const bits=[];
bits.push(driver==='Non determinato'
? 'Perché si muove non ancora identificato'
: `${driver} è il driver dominante`);
// Macro/Rates confirmation can be narrated here.
// Price is handled once, separately below, to avoid duplicate messages.
const nonPriceConfirm=(lead.confirm||'')
.split(' + ')
.filter(x=>x && x!=='Prezzo' && !x.startsWith('Nessuna'));
const nonPriceDiverge=(lead.diverge||'')
.split(' + ')
.filter(x=>x && x!=='Prezzo' && !x.startsWith('Nessuna'));
if(nonPriceConfirm.length){
bits.push(`${nonPriceConfirm.join(' + ')} conferma`);
}
if(nonPriceDiverge.length){
bits.push(`${nonPriceDiverge.join(' + ')} diverge`);
}
if(lead.lag && !lead.lag.startsWith('Nessun') && lead.lag!=='Non determinato'){
bits.push(`${lead.lag} è in ritardo`);
}
if(x?.confirmation==='PREZZO CONFERMA'){
bits.push('il prezzo conferma la struttura');
}else if(x?.confirmation==='PREZZO DIVERGE'){
bits.push('il prezzo diverge dalla struttura');
}else if(x?.confirmation==='PREZZO SENZA STRUTTURA'){
bits.push('il prezzo si muove, ma la struttura resta mista');
}else{
bits.push('il prezzo non dà ancora una conferma risolutiva');
}
if(lead.overlay && lead.overlay!=='Liquidità neutra'){
bits.push(lead.overlay.toLowerCase());
}
if(cot) bits.push(`${cot.toLowerCase()} sul posizionamento`);
return bits.join(' · ') + '.';
}

function currencyRiskOverlay(c){
const r=getCanonicalGlobalRisk();
if(!r || r.state==='WITHHELD') return '';
let relevance='contesto globale';
let detail=r.driver;
if(r.state==='SHOCK ENERGIA'){
if(c==='CAD'){
relevance='rilevante direttamente';
detail='Petrolio / terms of trade favorevoli se lo shock è supply-driven';
}else if(c==='JPY' || c==='EUR'){
relevance='rilevante direttamente';
detail='Energia importata / inflazione / crescita';
}else if(c==='USD'){
relevance='rilevante indirettamente';
detail='Inflazione / Treasury / aspettative Fed';
}else{
relevance='impatto secondario';
detail='Contesto globale, salvo shock molto ampio';
}
}else if(r.state==='RISK-OFF'){
if(['JPY','CHF'].includes(c)){
relevance='rilevante direttamente';
detail='Flussi difensivi / safe haven';
}else if(['AUD','NZD','CAD'].includes(c)){
relevance='rilevante direttamente';
detail='Sensibilità ciclica / risk appetite';
}else{
relevance='contesto globale';
}
}else if((r.affected||[]).includes(c)){
relevance='rilevante per questa valuta';
}
if(r.state==='RISCHIO CONTENUTO' && relevance==='contesto globale') return '';
return `<div class="globalRiskBadge ${riskClassFromState(r.state)}"><b>${r.state}</b> · ${relevance} · ${detail}</div>`;
}

function derivePairDriverStructure(pair){
const frame=derivePairDecisionFrame(pair);
if(!frame || frame.fav==='—') return {

codriver:null,
lead:'Non determinato',
reason:'Macro e Tassi/CB non identificano un vantaggio strutturale pulito.',
side:null,
source:'CANONICAL_PAIR_FRAME'
};
const [a,b]=pair.split('/');
const favSide=frame.fav===a?'A':frame.fav===b?'B':null;
const ms=frame.structural?.macro?.side;
const ps=frame.structural?.policy?.side;
let driver='Non determinato', codriver=null;
if(ms===favSide && ps===favSide){
driver='Macro + Rates / aspettative CB';
}else if(ms===favSide && ps==='TIE'){
driver='Macro';
}else if(ps===favSide && ms==='TIE'){
driver='Rates / aspettative CB';
}
return {
driver,
codriver,
lead:'Non determinato',
reason: driver==='Non determinato'
? 'I pilastri strutturali non sono abbastanza allineati per definire un driver unico.'
: `${driver} determina il vantaggio strutturale canonico di ${frame.fav}.`,
side:favSide,
source:'CANONICAL_PAIR_FRAME'
};
}
function currencyMomentumSignature(c){
const growth=blockDynamics(c,'Crescita');
const labour=blockDynamics(c,'Lavoro');
const rates=currencyDynamics(c)?.rates;
const out={positive:0,negative:0,turns:0,details:[],withheld:false};
const add=(name,dir,turn)=>{
if(dir>0){
out.positive++;
out.details.push(`${name}: migliora`);
}else if(dir<0){
out.negative++;
out.details.push(`${name}: peggiora`);
}
if(turn){
out.turns++;
out.details.push(`${name}: turning point`);
}
};
const growthReady=!!growth && growth.n>=2;
const labourReady=!!labour && labour.n>=2;
if(!growthReady || !labourReady || !rates){
out.withheld=true;
return out;
}
add('Crescita',growth.direction,growth.turning===true);
add('Lavoro',labour.direction,labour.turning===true);
add('Rates',rates.dir,rates.turn===true);
// Inflation is descriptive and Liquidity is an overlay.
// Neither enters the core relative-momentum comparison.
return out;
}
function pairAttentionLevel(pair){
const f=derivePairDecisionFrame(pair);
if(!f || f.quality==='MISTO / NON RISOLUTIVO' || f.quality==='NESSUN CONTRASTO'){
