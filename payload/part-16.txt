<script id="v491-forward-causal-layer-js">
(function(){
'use strict';
const LEDGER='validation/FORWARD_SNAPSHOT_LEDGER_V1_2026-10-07.json';
const PAYLOAD='validation/DASHBOARD_PAIR_RENDER_PAYLOAD_V1_2026-10-07.json';
let cache=null;
const stateLabel=s=>({TRANSMISSION_CLEAN:'CLEAN',TRANSMISSION_PARTIAL:'PARTIAL',TRANSMISSION_DIVERGENT:'DIVERGENT',TRANSMISSION_WITHHELD:'WITHHELD'}[s]||s||'—');
const stateCls=s=>String(s||'').replace('TRANSMISSION_','').toLowerCase();
function ensure(){
 const host=document.getElementById('forwarddesk'); if(!host) return null;
 let el=document.getElementById('fwdCausalV1');
 if(!el){el=document.createElement('div');el.id='fwdCausalV1';host.insertAdjacentElement('afterbegin',el);} return el;
}
async function load(){
 if(cache) return cache;
 const [lr,pr]=await Promise.all([fetch(LEDGER,{cache:'no-store'}),fetch(PAYLOAD,{cache:'no-store'})]);
 if(!lr.ok||!pr.ok) throw new Error('Nuovo forward ledger non disponibile');
 cache={ledger:await lr.json(),payload:await pr.json()}; return cache;
}
function outcomeCell(x){return x==null?'<span class="fwdCausalPending">IN ATTESA</span>':String(x);}
function pairRows(snap,payload){
 const buckets=[
  ['CLEAN robusti',snap.focus_buckets?.clean_robust||[]],
  ['Divergenze robuste',snap.focus_buckets?.divergent_robust||[]],
  ['Parziali robuste',snap.focus_buckets?.partial_robust||[]],
  ['Sensibili alla regola',snap.focus_buckets?.rule_sensitive||[]],
  ['Non classificati',snap.focus_buckets?.withheld||[]]
 ];
 return buckets.map(([title,pairs])=>{
  if(!pairs.length) return '';
  const rows=pairs.map(pair=>{const p=payload.pairs?.[pair]||{};return `<div class="fwdCausalRow"><b>${pair}</b><div><span class="fwdCausalState ${stateCls(p.state)}">${stateLabel(p.state)}</span>${p.robustness_badge==='RULE_SENSITIVE'?' · <span style="color:#fbbf24">RULE SENSITIVE</span>':''}</div><div>${outcomeCell(snap.forward_outcomes?.t_plus_5)}</div><div>${outcomeCell(snap.forward_outcomes?.t_plus_20)}</div><div>${outcomeCell(snap.forward_outcomes?.t_plus_60)}</div></div>`}).join('');
  return `<div class="fwdCausalBlock"><h4>${title}</h4><div class="fwdCausalRow head"><div>Cross</div><div>Stato congelato</div><div>5D</div><div>20D</div><div>60D</div></div>${rows}</div>`;
 }).join('');
}
async function render(){
 const root=ensure(); if(!root) return;
 root.innerHTML='<div class="fwdCausalPending">Caricamento nuovo Forward Test…</div>';
 try{
  const {ledger,payload}=await load(); const snap=ledger.snapshots?.[ledger.snapshots.length-1]; if(!snap) throw new Error('Snapshot forward assente');
  const c=snap.state_counts||{}, r=snap.robustness_counts||{};
  root.innerHTML=`<div class="fwdCausalHead"><div><h3>Forward Test · Causal Transmission V1</h3><div class="fwdCausalSub">Snapshot immutabile: Macro relativo → Banca centrale → Front-end/rates → Prezzo → stato di trasmissione. Il bias macro non viene mai ribaltato dai layer downstream.</div></div><div class="fwdCausalBadges"><span class="fwdCausalBadge">SNAPSHOT ${String(snap.captured_at||'').slice(0,10)}</span><span class="fwdCausalBadge">20D ORIZZONTE PRINCIPALE</span><span class="fwdCausalBadge">RESEARCH ONLY</span></div></div><div class="fwdCausalGrid"><div class="fwdCausalCard fwdCausalClean"><span>Clean</span><b>${c.TRANSMISSION_CLEAN??0}</b></div><div class="fwdCausalCard fwdCausalPartial"><span>Partial</span><b>${c.TRANSMISSION_PARTIAL??0}</b></div><div class="fwdCausalCard fwdCausalDiv"><span>Divergent</span><b>${c.TRANSMISSION_DIVERGENT??0}</b></div><div class="fwdCausalCard fwdCausalWithheld"><span>Withheld</span><b>${c.TRANSMISSION_WITHHELD??0}</b></div></div><div class="fwdCausalBadges"><span class="fwdCausalBadge">${r.ROBUST_ACROSS_RULES??0} robusti</span><span class="fwdCausalBadge">${r.RULE_SENSITIVE??0} rule-sensitive</span><span class="fwdCausalBadge">${snap.pair_count||0} cross</span></div>${pairRows(snap,payload)}<div class="fwdCausalNote"><b>Regola forward:</b> gli outcome a 5D / 20D / 60D vengono aggiunti solo quando maturano. Lo stato originale dello snapshot resta congelato; nessun tuning retroattivo. CHF/NZD mantengono un badge separato di evidenza storica incompleta.</div>`;
  window.__GMFQ_FORWARD_CAUSAL_V1__={status:'PASS',snapshot_id:snap.snapshot_id,pair_count:snap.pair_count};
 }catch(e){root.innerHTML=`<div class="fwdCausalErr">Forward Test V1 non caricato: ${e.message}</div>`;window.__GMFQ_FORWARD_CAUSAL_V1__={status:'FAIL',error:String(e.message||e)};}
}
const base=window.renderForwardMonitor;
if(typeof base==='function') window.renderForwardMonitor=function(){const out=base.apply(this,arguments);setTimeout(()=>render(),0);return out;};
window.renderForwardCausalV1=render;
if(document.readyState==='loading') document.addEventListener('DOMContentLoaded',()=>setTimeout(render,600)); else setTimeout(render,600);
})();
</script>
