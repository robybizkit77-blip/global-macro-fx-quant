function derivePairDecisionFrame(pair){
const [a,b]=pair.split('/');
const stateA=deriveCurrencyState(a);
const stateB=deriveCurrencyState(b);
const timingA=currencySignalMaturity(a);
const timingB=currencySignalMaturity(b);
// CANONICAL STRUCTURAL ENGINE:
// Only Macro + Rates/CB determine the relative structural advantage.
// Recent change (improves/worsens/turning), Price, COT and Liquidity
// are deliberately excluded from the structural direction.
const macroA=Number(stateA?.macro||0);
const macroB=Number(stateB?.macro||0);
const policyA=Number(stateA?.policy||0);
const policyB=Number(stateB?.policy||0);
const macroSide = macroA>macroB ? 'A' : macroB>macroA ? 'B' : 'TIE';
const policySide = policyA>policyB ? 'A' : policyB>policyA ? 'B' : 'TIE';
let fav='—';
let quality='MISTO / NON RISOLUTIVO';
let relation='Macro e Tassi/CB non separano abbastanza le due valute.';
if(macroSide!=='TIE' && macroSide===policySide){
fav=macroSide==='A'?a:b;
quality='CONTRASTO STRUTTURALE FORTE';
relation=`Macro e Tassi/CB favoriscono entrambi ${fav}: contrasto strutturale pieno.`;
}else if(macroSide!=='TIE' && policySide==='TIE'){
fav=macroSide==='A'?a:b;
quality='VANTAGGIO RELATIVO PARZIALE';
relation=`La Macro favorisce ${fav}, mentre Tassi/CB non separano le due valute: vantaggio presente ma incompleto.`;
}else if(policySide!=='TIE' && macroSide==='TIE'){
fav=policySide==='A'?a:b;
quality='VANTAGGIO RELATIVO PARZIALE';
relation=`Tassi/CB favoriscono ${fav}, mentre la Macro non separa le due valute: vantaggio presente ma incompleto.`;
}else if(macroSide!=='TIE' && policySide!=='TIE' && macroSide!==policySide){
quality='PILASTRI DIVISI';
relation='Macro e Tassi/CB favoriscono valute opposte: nessun vantaggio strutturale pulito.';
}else{
quality='NESSUN CONTRASTO';
}
const certified=typeof certifiedPairForMarket==='function'
? certifiedPairForMarket(pair)
: CERT53?.pairs?.[pair];
let priceSide=null;
if(certified){
const raw=certified.layer_directions?.price;
const side=normalizePairSide(raw);
priceSide=side==='A'?a:side==='B'?b:null;
}
let priceReading='PREZZO NON RISOLUTIVO';
let priceNote='Il Prezzo non aggiunge una lettura relativa abbastanza chiara.';
if(fav!=='—' && priceSide){
if(priceSide===fav){
priceReading='MERCATO GIÀ ALLINEATO';
priceNote=`Il Prezzo è già dalla parte di ${fav}. È conferma descrittiva, non prova di ulteriore continuazione.`;
}else{
priceReading='DIVERGENZA DA MONITORARE';
priceNote=`La struttura Macro + Tassi/CB favorisce ${fav}, ma il Prezzo favorisce ${priceSide}.`;
}
}
let missing='';
if(quality==='CONTRASTO STRUTTURALE FORTE'){
missing='Macro e Tassi/CB sono allineati sulla stessa valuta.';
}else if(quality==='VANTAGGIO RELATIVO PARZIALE'){
missing='Solo uno dei due pilastri strutturali separa chiaramente le valute.';
}else if(quality==='PILASTRI DIVISI'){
missing='Macro e Tassi/CB sono in conflitto: serve convergenza prima di promuovere il pair.';
}else{
missing='Serve maggiore differenziazione strutturale tra le due valute.';
}
const timing =
`${a}: ${timingA?.timing||'non determinato'} · ${b}: ${timingB?.timing||'non determinato'}`;
const watch =
priceReading==='DIVERGENZA DA MONITORARE'
? `${priceNote} Il Prezzo resta solo una verifica, non modifica la struttura.`
: priceReading==='MERCATO GIÀ ALLINEATO'
? `${priceNote} Evitare di confondere conferma con nuovo vantaggio strutturale.`
: `${priceNote} Priorità alla struttura Macro + Tassi/CB.`;
return {
fav,
quality,
relation,
timing,
priceReading,
priceNote,
missing,
watch,
structural:{
macro:{a:macroA,b:macroB,side:macroSide},
policy:{a:policyA,b:policyB,side:policySide}
}
};
}

function deriveCurrencyDrivers(c){

const macro=layerPolarity(d.macro);
const rates=policyFamilyPolarity(d);

// Macro = fundamental anchor.
// Rates/CB expectations = market translation.

// to claim which one is the primary driver.
// Liquidity remains an overlay and does not compete for the core driver.

function median(arr){
const x=arr.filter(Number.isFinite).slice().sort((a,b)=>a-b);
if(!x.length) return null;
const m=Math.floor(x.length/2);
return x.length%2?x[m]:(x[m-1]+x[m])/2;
}
function seriesDynamics(seriesObj){
if(!seriesObj?.values?.length) return null;
const vals=seriesObj.values.map(Number).filter(Number.isFinite);
if(vals.length<10) return null;
const n=vals.length;
const fast=Math.min(5, Math.floor(n/3));
const slow=Math.min(10, Math.floor(n/2));
const slope=(end,len)=>{
const st=end-len+1;
if(st<0) return null;
const a=vals[st], b=vals[end];
return (b-a)/(len-1);
};
const recent=slope(n-1,fast);
const prior=slope(n-fast,fast);
if(recent==null || prior==null) return null;
const diffs=[];
for(let i=Math.max(1,n-52);i<n;i++) diffs.push(Math.abs(vals[i]-vals[i-1]));
const typical=median(diffs)||0;
const dir=Math.abs(recent)<(typical*0.15)?0:Math.sign(recent);
const speed=typical>0?Math.abs(recent)/typical:null;
const accel=recent-prior;
const accelDir=Math.abs(accel)<(typical*0.15)?0:Math.sign(accel);
const turn=(Math.sign(prior)!==0 && Math.sign(recent)!==0 && Math.sign(prior)!==Math.sign(recent));
return {recent,prior,dir,speed,accel,accelDir,turn};
}

function robustScaleDiffs(vals){
const diffs=[];
for(let i=1;i<vals.length;i++){
const d=Number(vals[i])-Number(vals[i-1]);
if(isFinite(d)) diffs.push(Math.abs(d));
}
const m=median(diffs);
return (m && isFinite(m) && m>0)?m:null;
}
function macroSeriesPolarity(category,label){
const l=String(label||'').toLowerCase();
if(category==='Lavoro'){
if(l.includes('disoccup')) return -1;
if(l.includes('sussidi disoccupazione')) return -1;
if(l.includes('claims')) return -1;
if(l.includes('costo unitario')) return 0; // inflation/labour-cost signal, not clean labour-health direction
return 1;
}
if(category==='Crescita'){
if(l.includes('tasso di risparmio')) return 0; // ambiguous for contemporaneous demand
return 1;
}
if(category==='Inflazione'){
return 1; // direction = inflation pressure rising/falling, NOT FX-positive/negative
}
if(category==='Liquidità'){
return 1; // quantity direction only; not automatically FX-positive
}
return 0;
}
function normalizedSeriesImpulse(x,category){
const vals=(x?.values||[]).map(Number).filter(Number.isFinite);
if(vals.length<8) return null;
const pol=macroSeriesPolarity(category,x.label);
if(pol===0) return null;
const scale=robustScaleDiffs(vals.slice(-80));
if(!scale) return null;
const d0=(vals.at(-1)-vals.at(-2))*pol/scale;
const d1=(vals.at(-2)-vals.at(-3))*pol/scale;
return {current:d0,previous:d1,label:x.label};
}

function blockDynamics(c,category){
const list=(S?.[c]||[]).filter(x=>x.category===category);
const impulses=list.map(x=>normalizedSeriesImpulse(x,category)).filter(Boolean);
if(!impulses.length) return null;
const current=medianNumber(impulses.map(x=>x.current));
const previous=medianNumber(impulses.map(x=>x.previous));
if(current==null || previous==null) return null;
const direction=Math.abs(current)<0.20?0:Math.sign(current);
const acceleration=current-previous;
const accelDir=Math.abs(acceleration)<0.20?0:Math.sign(acceleration);
const turning=(Math.sign(previous)!==0 && direction!==0 && Math.sign(previous)!==direction);
return {
category,
direction,
speed:Math.abs(current),
acceleration,
accelDir,
turning,
n:impulses.length
};
}

function currencyEarlyShift(c){
const imp=currencyImprovementDiagnostic(c);
const signals=[];
const growth=interpretStateVsDynamics(c,'Crescita');
const labour=interpretStateVsDynamics(c,'Lavoro');
const rates=currencyDynamics(c)?.rates;
if(growth && growth.label!=='Withheld' && growth.label!=='Stabile'){
signals.push({cat:'Crescita',label:growth.label,note:growth.note,type:'core'});
}
if(labour && labour.label!=='Withheld' && labour.label!=='Stabile'){
signals.push({cat:'Lavoro',label:labour.label,note:labour.note,type:'core'});
}
if(rates){
const rateLabel =
rates.turn ? 'Turning point Rates' :
rates.dir>0 ? 'Rates migliorano' :
rates.dir<0 ? 'Rates peggiorano' :
'Rates stabili';
if(rateLabel!=='Rates stabili'){
signals.push({cat:'Rates',label:rateLabel,note:'dinamica front-end',type:'core'});
}
}
if(imp.state==='NON DETERMINATO'){
return {c,state:'WITHHELD',cls:'earlyNone',signals,active:false};
}
if(imp.state==='EARLY SHIFT POSITIVO'){
return {c,state:'Warning anticipato positivo · Macro non conferma',cls:'earlyWarn',signals,active:true};
}
if(imp.state==='EARLY SHIFT NEGATIVO'){
return {c,state:'Warning anticipato negativo · Macro non conferma',cls:'earlyWarn',signals,active:true};
}
if(imp.state==='TURNING POINT'){
return {c,state:'Turning Point',cls:'earlyWarn',signals,active:true};
}
if(imp.state==='MIGLIORA'){
return {c,state:'Miglioramento confermato',cls:'earlyUp',signals,active:true};
}
if(imp.state==='PEGGIORA'){
return {c,state:'Deterioramento confermato',cls:'earlyWarn',signals,active:true};
}
return {c,state:'Nessun early shift',cls:'earlyNone',signals,active:false};
}

function canonicalViewQa(){
const issues=[];
if(typeof selectedCcy!=='undefined' && MODEL_STATE.selectedCurrency!==selectedCcy){
issues.push('selectedCurrency fuori sync');
}
const c=MODEL_STATE.selectedCurrency;
if(!MODEL_STATE.currency?.[c]) issues.push('selected currency state mancante');

function canonicalConsistencyQa(){
const issues=[];
const ccys=['USD','EUR','GBP','JPY','CHF','CAD','AUD','NZD'];
if(!MODEL_STATE.updatedAt) issues.push('state non inizializzato');
if(Object.keys(MODEL_STATE.currency||{}).length!==8) issues.push('currency state incompleto');
if(Object.keys(MODEL_STATE.pairs||{}).length!==28) issues.push('pair state incompleto');
ccys.forEach(c=>{
const x=MODEL_STATE.currency?.[c];
if(!x) issues.push(`${c}: stato mancante`);
else{
if(!x.driver) issues.push(`${c}: driver mancante`);
if(!x.lead) issues.push(`${c}: lead mancante`);
if(!x.regime) issues.push(`${c}: regime mancante`);
}
});
const selected=MODEL_STATE.selectedCurrency;
if(selected && typeof selectedCcy!=='undefined' && selected!==selectedCcy){
issues.push(`selected currency mismatch ${selected}/${selectedCcy}`);
}
return {
ok:issues.length===0,
issues,
currencies:Object.keys(MODEL_STATE.currency||{}).length,
pairs:Object.keys(MODEL_STATE.pairs||{}).length,
risk:MODEL_STATE.globalRisk?.state || 'WITHHELD'
};
}

}
function v461PriceExtension(c){
  if(c==='USD') return {state:'BASE',pct:null,move:null,text:'USD è la base del confronto.'};
  const full=ccySeries(c,D.prices.length);
  const v=(full&&full.values)||[];
  const win=22;
  if(v.length<win+30) return {state:'WITHHELD',pct:null,move:null,text:'Storico prezzo insufficiente.'};
  const hist=[];
  for(let i=win;i<v.length;i++){
    const a=Number(v[i-win]),b=Number(v[i]);
    if(!Number.isFinite(a)||!Number.isFinite(b)||a===0) continue;
    hist.push((b/a-1)*100);
  }
  if(hist.length<80) return {state:'WITHHELD',pct:null,move:null,text:'Storico prezzo insufficiente.'};
  const cur=hist.at(-1), sorted=[...hist].sort((a,b)=>a-b);
  let le=0; for(const x of sorted) if(x<=cur) le++;
  const pct=100*le/sorted.length;
  const absTail=Math.max(pct,100-pct);
  if(absTail>=90) return {state:'EXTENDED',pct,move:cur,text:`Movimento 1M molto avanzato rispetto allo storico · P${pct.toFixed(0)}.`};
  if(absTail>=80) return {state:'ADVANCED',pct,move:cur,text:`Movimento 1M già avanzato rispetto allo storico · P${pct.toFixed(0)}.`};
  return {state:'NORMAL',pct,move:cur,text:`Movimento 1M non particolarmente esteso · P${pct.toFixed(0)}.`};
}
function v461OperationalRead(c){
  const st=deriveCurrencyState(c)||{};
  const d=currencyIntelData?.[c]||{};
  const lead=deriveLeadStructure(c)||{};
  const plan=deriveCurrencyActionPlan(c)||{};
  const ext=v461PriceExtension(c);
  const driver=deriveCurrencyDriver(c)||'Non determinato';
  let second=lead.confirm&&lead.confirm!=='Nessuna conferma chiara'?lead.confirm:'Nessuna conferma secondaria forte';
  let brake='Nessun freno dominante';
  if(st.confirmation==='PREZZO DIVERGE') brake='Prezzo: non conferma ancora la struttura';
  else if(st.cotRisk) brake=`COT: ${st.cotRisk.toLowerCase()}`;
  else if(st.liquidityOverlay==='LIQUIDITÀ CONTRARIA') brake='Liquidità contraria';
  let priced='Il prezzo non sta ancora incorporando in modo pulito la storia strutturale.';
  if(st.confirmation==='PREZZO CONFERMA') priced='Il prezzo è già allineato a Macro + Tassi/CB: una parte della storia è già riflessa.';
  else if(st.confirmation==='PREZZO DIVERGE') priced='La struttura e il prezzo sono ancora separati: la storia non è pienamente prezzata.';
  let surprise='Una sorpresa sarebbe un movimento netto di Macro o Tassi/CB contro il driver dominante.';
  if(st.base==='MISTA') surprise='La vera sorpresa sarebbe vedere Macro e Tassi/CB finalmente convergere nella stessa direzione.';
  const profit=ext.state==='EXTENDED'||ext.state==='ADVANCED'
    ? ext.text+' Il rischio di presa di profitto è quindi più alto.'
    : ext.state==='WITHHELD'
      ? 'WITHHELD: storico insufficiente per valutare l’estensione.'
      : ext.text+' Nessun segnale meccanico di eccesso.';
  return {
    structural:v461StateLabel(c),
    change:v461ChangeLabel(c),
    price:v461PriceLabel(c),
    driver,
    second,
    brake,
    missing:plan.strengthen||'Serve maggiore coerenza tra Macro, Tassi/CB e Prezzo.',
    priced,
    surprise,
    profit,
    next:d.watch||plan.watch||'Controlla i prossimi dati macro e il front-end rates.'
  };
}
