(function(){
  'use strict';

  function selectedPair(){
    const sel=document.querySelector('#pairSelect');
    if(sel?.value) return sel.value;
    const t=document.querySelector('#v226PairTitle')?.textContent||'';
    return /^([A-Z]{3}\/[A-Z]{3})$/.test(t.trim())?t.trim():null;
  }

  function sideText(pair,fav){
    const [base,quote]=String(pair||'').split('/');
    if(!fav||fav==='—') return {txt:'NESSUN VANTAGGIO',cls:'neu'};
    if(fav===base) return {txt:`LONG ${pair}`,cls:'pos'};
    if(fav===quote) return {txt:`SHORT ${pair}`,cls:'neg'};
    return {txt:'MISTA',cls:'warn'};
  }

  function direction(pair,frame){
    let dyn=null, trans=null;
    try{ dyn=window.derivePairRelativeDynamics?.(pair)||null; }catch(e){}
    try{ trans=window.pairTransitionState?.(pair)||null; }catch(e){}
    if(!frame?.fav || frame.fav==='—') return {txt:'DIREZIONE INCERTA',cls:'warn'};
    if(trans?.state==='DINAMICA RELATIVA ALLINEATA') return {txt:'SI RAFFORZA ↗',cls:'pos'};
    if(trans?.state==='DIVERGENZA RELATIVA DA MONITORARE') return {txt:'SI INDEBOLISCE ↘',cls:'neg'};
    if(dyn?.state==='TURNING POINT ASIMMETRICO') return {txt:'POSSIBILE SVOLTA',cls:'warn'};
    return {txt:'STABILE →',cls:'neu'};
  }

  function sideLabel(pair,side){
    const [a,b]=pair.split('/');
    if(side==='A') return a;
    if(side==='B') return b;
    return 'nessuna delle due';
  }

  function model(pair){
    let frame={}, setup={};
    try{ frame=window.derivePairDecisionFrame?.(pair)||{}; }catch(e){}
    try{ setup=window.derivePairSetupState?.(pair)||{}; }catch(e){}
    const structure=sideText(pair,frame.fav), dir=direction(pair,frame);
    const [a,b]=pair.split('/');

    let title,text;
    if(frame.fav && frame.fav!=='—'){
      const other=frame.fav===a?b:a;
      if(frame.quality==='CONTRASTO STRUTTURALE FORTE'){
        title=`${frame.fav} ha un vantaggio strutturale chiaro su ${other}`;
        text=`Macro e Tassi/CB puntano entrambi verso ${frame.fav}. Questo è il tipo di contrasto relativo più pulito: il pair ha una direzione strutturale, mentre prezzo e COT servono soprattutto a capire timing e rischio.`;
      }else{
        title=`${frame.fav} è avanti, ma il vantaggio non è ancora completo`;
        text=`Solo uno tra Macro e Tassi/CB separa chiaramente le due valute. La storia esiste, ma manca ancora una seconda conferma strutturale prima di considerarla davvero pulita.`;
      }
    }else{
      title=`${pair} non ha ancora un vantaggio strutturale chiaro`;
      text=`Macro e Tassi/CB non separano abbastanza le due valute oppure puntano in direzioni opposte. In questa situazione è meglio non forzare una view sul pair.`;
    }

    const macroSide=sideLabel(pair,frame?.structural?.macro?.side);
    const policySide=sideLabel(pair,frame?.structural?.policy?.side);
    const priceTxt=frame?.priceReading==='MERCATO GIÀ ALLINEATO'?'Il prezzo conferma':frame?.priceReading==='DIVERGENZA DA MONITORARE'?'Il prezzo diverge':'Il prezzo non è risolutivo';
    const priceCls=frame?.priceReading==='MERCATO GIÀ ALLINEATO'?'pos':frame?.priceReading==='DIVERGENZA DA MONITORARE'?'warn':'neu';
    const drivers=[
      {label:'Macro',txt:macroSide==='nessuna delle due'?'Non separa le due valute':`Favorisce ${macroSide}`,cls:macroSide===frame.fav?'pos':macroSide==='nessuna delle due'?'neu':'warn'},
      {label:'Tassi / CB',txt:policySide==='nessuna delle due'?'Non separano le due valute':`Favoriscono ${policySide}`,cls:policySide===frame.fav?'pos':policySide==='nessuna delle due'?'neu':'warn'},
      {label:'Prezzo',txt:priceTxt,cls:priceCls}
    ];

    let confirm,reverse;
    if(frame.fav && frame.fav!=='—'){
      confirm=frame.quality==='CONTRASTO STRUTTURALE FORTE'
        ? `Il vantaggio si rafforza se Macro e Tassi/CB restano allineati su ${frame.fav} e il prezzo continua o inizia a confermare.`
        : `Il vantaggio diventa più credibile quando anche il pilastro oggi neutro si allinea a ${frame.fav}.`;
      reverse=`Dobbiamo rivedere la view se il vantaggio Macro/Tassi si perde, se i due pilastri iniziano a favorire valute opposte o se la divergenza di prezzo persiste mentre il core si deteriora.`;
    }else{
      confirm='Serve che Macro e Tassi/CB inizino a favorire la stessa valuta. Finché non succede, il pair resta da aspettare.';
      reverse='Se Macro e Tassi/CB continuano a essere divisi, non c’è una view strutturale da invalidare: semplicemente non c’è ancora un setup.';
    }

    const quality=setup?.quality||frame?.quality||'—';
    const conclusion=frame.fav&&frame.fav!=='—'
      ? `Conclusione: ${frame.fav} è la valuta favorita nel confronto. Qualità del setup: ${quality}. COT e liquidità restano contesto/rischio; il prezzo conferma o diverge, ma non decide la struttura.`
      : `Conclusione: nessun vantaggio strutturale pulito su ${pair}. Aspetta una differenza più chiara tra le due valute.`;

    return {structure,dir,title,text,drivers,confirm,reverse,conclusion,frame,setup};
  }

  function ensureUi(){

      head.insertAdjacentElement('afterend',ticks);
    }
    if(!document.getElementById('gmfqPairTutor')){
      const box=document.createElement('div');
      box.id='gmfqPairTutor'; box.className='gmfqPairTutor';
      box.innerHTML='<div class="lab">COSA STA SUCCEDENDO?</div><div id="gmfqPairTitle" class="ttl">—</div><div id="gmfqPairText" class="txt">—</div><div id="gmfqPairDrivers" class="drivers"></div><div class="watch"><div class="good"><small>COSA RAFFORZA QUESTA LETTURA?</small><b id="gmfqPairConfirm">—</b></div><div class="bad"><small>COSA LA METTE IN DUBBIO?</small><b id="gmfqPairReverse">—</b></div></div><div class="confirm"><b>CONCLUSIONE</b><div id="gmfqPairConclusion">—</div></div>';
