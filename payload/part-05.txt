function theoreticalScenarios(e){
const n=(e.event||'').toLowerCase();
const normal=(strong,weak)=>({strong,weak});
const hawkish=(ccy)=>normal(
`Più forte / sopra attese → pricing più hawkish → ${ccy} teoricamente favorito`,
`Più debole / sotto attese → pricing più dovish → ${ccy} teoricamente sfavorito`
);
if(n.includes('rate statement') || n.includes('press conference') || n.includes('speaks') || n.includes('minutes') || n.includes('summary of opinions')){
return normal(
`Tono più hawkish del previsto → ${e.ccy} teoricamente favorito`,
`Tono più dovish del previsto → ${e.ccy} teoricamente sfavorito`
);
}
if(n.includes('bond auction')){
return normal(
'Domanda più forte / rendimento più contenuto → minore stress sul debito → supporto EUR',
'Domanda più debole / rendimento più alto → maggiore stress sul debito → pressione EUR'
);
}
if(n.includes('unemployment rate')){
return normal(
`Disoccupazione più bassa del previsto → mercato del lavoro più forte → ${e.ccy} teoricamente favorito`,
`Disoccupazione più alta del previsto → mercato del lavoro più debole → ${e.ccy} teoricamente sfavorito`
);
}
if(n.includes('unemployment change')){
return normal(
`Variazione occupazionale migliore / disoccupazione più bassa → ${e.ccy} teoricamente favorito`,
`Variazione occupazionale peggiore / disoccupazione più alta → ${e.ccy} teoricamente sfavorito`
);
}
if(n.includes('cpi') || n.includes('pce') || n.includes('sppi') || n.includes('import prices')){
return hawkish(e.ccy);
}
if(n.includes('gdp') || n.includes('pmi') || n.includes('retail') || n.includes('consumer spending') ||
n.includes('durable goods') || n.includes('industrial production') || n.includes('business confidence') ||
n.includes('consumer confidence') || n.includes('jolts') || n.includes('jobless claims') ||
n.includes('trade balance') || n.includes('m3 money supply') || n.includes('tankan')){
if(n.includes('jobless claims')){
return normal(
`Claims più bassi del previsto → lavoro più forte → ${e.ccy} teoricamente favorito`,
`Claims più alti del previsto → lavoro più debole → ${e.ccy} teoricamente sfavorito`
);
}
return normal(
`Dato più forte / sopra attese → crescita o domanda più forti → ${e.ccy} teoricamente favorito`,
`Dato più debole / sotto attese → crescita o domanda più deboli → ${e.ccy} teoricamente sfavorito`
);
}
return normal(
`Scenario più forte / più hawkish → ${e.ccy} teoricamente favorito`,
`Scenario più debole / più dovish → ${e.ccy} teoricamente sfavorito`
);
}
