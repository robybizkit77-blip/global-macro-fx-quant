function v232ParseDate(d){

function v232Range(series,tf){

if(tf==='MAX' || !dates.length) return {dates,vals};
const parsed=dates.map(v232ParseDate);
const last=[...parsed].reverse().find(Boolean);
if(!last) return {dates,vals};
const months={'1M':1,'3M':3,'6M':6,'1Y':12,'2Y':24,'5Y':60}[tf]||9999;

return {dates:dates.slice(idx>=0?idx:0),vals:vals.slice(idx>=0?idx:0)};
}

function v474MacroSimpleModel(c,snap){
  const head=v232Headline(c,snap);
  const stateMap={
    pos:{label:'Solida',cls:'pos'},
    neg:{label:'Debole',cls:'neg'},
    warn:{label:'Mista',cls:'warn'},
    neu:{label:'Non chiara',cls:'neu'}
  };
  const overall=stateMap[head.cls]||stateMap.neu;

  const nonInfl=snap.sectors.filter(x=>!/inflaz|prezzi/i.test(String(x.label||'')));
  const sig=nonInfl.filter(x=>x.trend && x.trend.significant);
  const improving=sig.filter(x=>x.trend.cls==='pos').length;
  const worsening=sig.filter(x=>x.trend.cls==='neg').length;
  const turning=sig.filter(x=>x.trend.cls==='warn').length;

  let change={label:'Stabile',cls:'neu'};
  // A material turning signal must not be hidden by a simple positive-vs-negative count.
  // Directional labels require a clean directional majority; otherwise keep the state mixed/turning.
  if(turning>0) change={label:'Possibile svolta / quadro misto',cls:'warn'};
  else if(improving>worsening && improving>=1) change={label:'Migliora',cls:'pos'};
  else if(worsening>improving && worsening>=1) change={label:'Peggiora',cls:'neg'};
  else if(sig.length) change={label:'Possibile svolta / quadro misto',cls:'warn'};

  const strong=snap.strength?.label||'un’area di tenuta';
  const weak=snap.weakness?.label||'un’area di fragilità';
  const changing=snap.changing?.label||weak;
  const changeWhy=snap.changing?.trend?.why||'Non emerge ancora un cambiamento netto.';

  let simple;
  if(overall.cls==='pos'){
    simple=`L’economia di ${v232CountryLabel(c)} nel complesso regge bene. ${strong} è il principale punto di forza, mentre ${weak} resta l’area da controllare.`;
  }else if(overall.cls==='neg'){
    simple=`L’economia di ${v232CountryLabel(c)} sta mostrando debolezza in più aree. ${weak} è il freno principale; ${strong} è ciò che per ora evita un quadro ancora più fragile.`;
  }else{
    simple=`L’economia di ${v232CountryLabel(c)} manda segnali contrastanti: alcune aree tengono, altre frenano. ${strong} aiuta, mentre ${weak} resta la fragilità principale.`;
  }

  let why=`Tiene soprattutto ${String(strong).toLowerCase()}; frena soprattutto ${String(weak).toLowerCase()}. ${changeWhy}`;

  let confirm, invalidate;
  if(overall.cls==='pos'){
    confirm=`La lettura si rafforza se ${String(strong).toLowerCase()} resta solido e ${String(weak).toLowerCase()} smette di peggiorare o migliora.`;
    invalidate=`La lettura si indebolisce se ${String(weak).toLowerCase()} peggiora ancora e il punto di forza in ${String(strong).toLowerCase()} comincia a cedere.`;
  }else if(overall.cls==='neg'){
    confirm=`La lettura si rafforza se ${String(weak).toLowerCase()} resta debole o peggiora e anche ${String(changing).toLowerCase()} conferma il deterioramento.`;
    invalidate=`La lettura viene messa in dubbio se ${String(weak).toLowerCase()} migliora in modo chiaro e ${String(strong).toLowerCase()} continua a reggere.`;
  }else{
    confirm=`Il quadro misto si conferma se ${String(strong).toLowerCase()} continua a tenere ma ${String(weak).toLowerCase()} resta fragile, senza una direzione comune.`;
    invalidate=`Il quadro smette di essere misto se più settori iniziano a muoversi nella stessa direzione: miglioramento diffuso oppure deterioramento diffuso.`;
  }

  return {overall,change,simple,why,confirm,invalidate};
}

    reverse=`Se ${String(weak?.label||'la parte più debole').toLowerCase()} migliora in modo chiaro e ${String(strong?.label||'il punto di forza').toLowerCase()} continua a reggere, dobbiamo rivedere la lettura negativa.`;

    reverse=`Se ${String(strong?.label||'il punto di forza').toLowerCase()} comincia a cedere e ${String(weak?.label||'la parte fragile').toLowerCase()} peggiora, dobbiamo ridurre la fiducia nel quadro positivo.`;

  }
  return {title,text,change,drivers,confirm,reverse,fx};
}

function v476Direction(x){
  const label=String(x?.label||'');
  const t=String(x?.trend?.txt||'').toLowerCase();
  const cls=String(x?.trend?.cls||'neu');

  if(/inflaz|salari/i.test(label)){
    if(/aumento|accelerazione|sale/.test(t)) return {txt:'Pressioni in aumento',cls:'v476Warn'};
    if(/disinflazione|calo|dimin/.test(t)) return {txt:'Pressioni in calo',cls:'v476Info'};
    if(/svolta|regime/.test(t)) return {txt:'Possibile cambio di direzione',cls:'v476Warn'};
    return {txt:'Abbastanza stabile',cls:'v476Neu'};
  }

  if(/positiva|migliora|ripresa|allentamento/.test(t)) return {txt:'Sta migliorando',cls:'v476Pos'};
  if(/negativa|peggiora|deterior|raffreddamento|restritt|irrigid/.test(t)) return {txt:'Sta peggiorando',cls:'v476Neg'};
  if(/svolta|monitorare|misto|cambiando/.test(t)) return {txt:'Direzione ancora incerta',cls:'v476Warn'};
  if(cls==='pos') return {txt:'Sta migliorando',cls:'v476Pos'};
  if(cls==='neg') return {txt:'Sta peggiorando',cls:'v476Neg'};
  if(cls==='info') return {txt:'Sta cambiando',cls:'v476Info'};
  return {txt:'Stabile',cls:'v476Neu'};
}
